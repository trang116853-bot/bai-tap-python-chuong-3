import sys

def tim_cung_hoang_dao(d, m):
    if (m == "Mar" and d >= 21) or (m == "Apr" and d <= 20):
        return "ARIES"
    elif (m == "Apr" and d >= 21) or (m == "May" and d <= 20):
        return "TAURUS"
    elif (m == "May" and d >= 21) or (m == "Jun" and d <= 20):
        return "GEMINI"
    elif (m == "Jun" and d >= 21) or (m == "Jul" and d <= 22):
        return "CANCER"
    elif (m == "Jul" and d >= 23) or (m == "Aug" and d <= 22):
        return "LEO"
    elif (m == "Aug" and d >= 23) or (m == "Sep" and d <= 22):
        return "VIRGO"
    elif (m == "Sep" and d >= 23) or (m == "Oct" and d <= 22):
        return "LIBRA"
    elif (m == "Oct" and d >= 23) or (m == "Nov" and d <= 22):
        return "SCORPIO"
    elif (m == "Nov" and d >= 23) or (m == "Dec" and d <= 21):
        return "SAGITTARIUS"
    elif (m == "Dec" and d >= 22) or (m == "Jan" and d <= 19):
        return "CAPRICORN"
    elif (m == "Jan" and d >= 20) or (m == "Feb" and d <= 19):
        return "AQUARIUS"
    elif (m == "Feb" and d >= 20) or (m == "Mar" and d <= 20):
        return "PISCES"

input_data = sys.stdin.read().split()
if input_data:
    T = int(input_data[0])
    idx = 1
    for _ in range(T):
        d = int(input_data[idx])
        m = input_data[idx + 1]
        idx += 2
        print(tim_cung_hoang_dao(d, m))
